\begin{abstract}
This letter investigates a reconfigurable intelligent surface (RIS)-assisted
multiple-input single-output (MISO) integrated sensing and communication (ISAC)
system, where a base station (BS) simultaneously serves a near and a far user
via non-orthogonal multiple access (NOMA) while illuminating a sensing target.
We maximize a weighted communication-and-sensing throughput by jointly
optimizing the NOMA power-allocation coefficients and the RIS phase shifts,
subject to an aggregate communication QoS floor, individual near/far-user QoS
floors, and a sensing QoS constraint. To tackle the resulting non-convex problem,
a model-agnostic meta-learning (MAML)-based deep-learning algorithm is proposed.
A lightweight network predicts the low-dimensional power split, while an inner
optimizer adapts the high-dimensional RIS phase vector for each channel
realization. A first-order meta-update then trains the power map using the
phase-adapted channels, avoiding the unstable higher-order derivatives observed
in this high-SINR setting. Simulation results show
stable convergence, a strong RIS-aperture gain, and consistent improvements
against OMA, no-RIS, and partially optimized baselines.
\end{abstract}

\begin{IEEEkeywords}
Reconfigurable intelligent surface (RIS), integrated sensing and communication
(ISAC), non-orthogonal multiple access (NOMA), deep learning, meta-learning
(MAML), MISO.
\end{IEEEkeywords}

\section{Introduction}
\IEEEPARstart{I}{ntegrated} sensing and communication (ISAC) allows sensing and
data transmission to share spectrum, antennas, and RF hardware~\cite{liu2022isac}.
Power-domain NOMA improves spectral efficiency by superposing users and using
successive interference cancellation (SIC)~\cite{ding2017noma,sun2024noma},
while RISs can reconfigure blocked or weak propagation paths with programmable
passive elements~\cite{basar2019ris,wu2020ris}.

RIS-assisted ISAC has been studied through joint active/passive beamforming
\cite{wang2025hybrid}, CRB-oriented block-coordinate optimization
\cite{lyu2024crb}, and RIS-assisted rate-splitting~\cite{ke2025rsma}. For
RIS-NOMA,~\cite{zou2023ml} showed that learning/meta-learning can jointly adapt
resource allocation and RIS phases. More generally, MAML learns an
initialization that can adapt rapidly to a new task~\cite{maml,hospedales2022meta},
which is attractive here because the three power coefficients have reusable
structure across channels whereas the $N$ RIS phases are strongly
channel-specific.

The contributions are threefold: 1)~we formulate joint NOMA power and RIS-phase
optimization with cluster, per-user, and sensing QoS constraints; 2)~we develop
a first-order MAML bilevel solver in which a compact network predicts the power
split and a short inner loop adapts the RIS phases; and 3)~we analyze convergence
limitations and complexity and compare RIS/no-RIS and NOMA/OMA designs on
held-out channels.

\subsection{System Model}
As shown in Fig.~\ref{fig:system}, an $M$-antenna BS serves near/far users and
a sensing target through an $N$-element RIS. Element $i$ is modeled as
\begin{equation}
v_i=\rho_i e^{j\phi_i},\qquad 0\leq\rho_i\leq1,\quad \phi_i\in[0,2\pi),
\end{equation}
with $\bPhi=\operatorname{diag}(v_1,\ldots,v_N)$. The ideal passive-RIS
setting used in simulation is given in Table~\ref{tab:system}.

\subsubsection{Channel Model}
Let $\bH_{BR}\in\Cset^{N\times M}$ denote the BS--RIS channel and let
$\bH_{RB}$ denote the reciprocal RIS--BS channel. For user
$k\in\{n,f\}$, $\bh_{RD_k}\in\Cset^N$ denotes the RIS--user channel. The
effective user channel is
\begin{equation}
\bh_k=\bH_{RB}\bPhi\bh_{RD_k}\in\Cset^M. \label{eq:cascade}
\end{equation}
Direct BS--user paths are blocked in the RIS-assisted data set. With
$\bh_{RT}$ and $\bh_{TR}$ denoting the RIS--target and target--RIS links, the
rank-one round-trip response is
\begin{equation}
\bG_T=\sqrt{\beta_T}
(\bH_{RB}\bPhi\bh_{TR})
(\bh_{RT}\herm\bPhi\bH_{BR}), \label{eq:sensingchannel}
\end{equation}
where $\beta_T$ is the target/path-gain coefficient; all numerical channel
settings are listed in Section~IV.

\subsubsection{NOMA Communication and Sensing Model}
The transmitted signal is
\begin{equation}
\bm{x}=\left(\sqrt{a_nP}s_n+\sqrt{a_fP}s_f\right)\bw_c
       +\sqrt{a_TP}\bw_Ts_T, \label{eq:tx}
\end{equation}
where $\bm a=[a_n,a_f,a_T]$ is a nonnegative power split with
$a_n+a_f+a_T=1$. The unit-norm communication beam is
\begin{equation}
\bw_c=
\frac{\mu_f\bh_f+\mu_n\bh_n}
{\|\mu_f\bh_f+\mu_n\bh_n\|},\qquad
\mu_f,\mu_n\ge0,\quad \mu_f+\mu_n=1, \label{eq:wc}
\end{equation}
where $\mu_f$ and $\mu_n$ control the far/near steering emphasis; both users
share $\bw_c$ and are separated by SIC~\cite{ding2017noma,zou2023ml}.

For sensing, define $\bH_u=[\bh_n,\bh_f]$ and the leakage-aware matrix
\begin{equation}
\bA_T=\bG_T\herm\bG_T-\lambda_L\bH_u\bH_u\herm. \label{eq:wTmat}
\end{equation}
The sensing beam is the normalized dominant eigenvector of $\bA_T$; hence
$\lambda_L$ controls user leakage~\cite{wang2025hybrid,lyu2024crb}.

With $g_{kj}=|\bh_k\herm\bw_j|^2$, the SINRs used by the implementation are
\begin{align}
\gamma_f &= \frac{a_fg_{ff}P}{a_ng_{ff}P+a_Tg_{fT}P+\sigma^2}, \label{eq:gf}\\
\gamma_n &= \frac{a_ng_{nn}P}{a_Tg_{nT}P+\sigma^2}, \label{eq:gn}\\
\gamma_s &= \frac{a_T|\bm u_T\herm\bG_T\bw_T|^2P}{\sigma^2}, \label{eq:gs}
\end{align}
where $\sigma^2$ is noise power, $\bm u_T$ is the sensing combiner, and
$R_k=\log_2(1+\gamma_k)$. The weighted objective is
\begin{equation}
\Rdl=\omega_c(R_n+R_f)+\omega_sR_s,\qquad \omega_c+\omega_s=1.
\label{eq:rdl}
\end{equation}
Here $R_n+R_f$ is the NOMA-cluster spectral efficiency and
$\omega_c,\omega_s$ set the communication/sensing priority.

\subsection{Problem Formulation}\label{sec:prob}
We jointly optimize the power split $\bm{a}$ and RIS phases
$\bm{\phi}=[\phi_1,\ldots,\phi_N]$. The communication sum rate is retained as a
cluster-level QoS measure, while individual user constraints are added in the
same spirit as the per-device QoS constraint in (9b) of~\cite{zou2023ml}. The
resulting problem is
\begin{subequations}\label{eq:prob}
\begin{align}
\text{(P1)}\quad \max_{\bm{a},\bm{\phi}}\quad & \Rdl \label{eq:obj}\\
\text{s.t.}\quad
& R_n+R_f\geq R_{\mathrm{th},c}, \label{eq:qosc}\\
& R_k\geq R_{\mathrm{th},k},\quad k\in\{n,f\}, \label{eq:qosu}\\
& R_s\geq R_{\mathrm{th},s}, \label{eq:qoss}\\
& a_n+a_f+a_T=1,\quad a_n,a_f,a_T\geq0, \label{eq:simplex}\\
& |e^{j\phi_i}|=1,\quad i=1,\ldots,N. \label{eq:unit}
\end{align}
\end{subequations}
Constraint~\eqref{eq:qosc} preserves cluster throughput, while
\eqref{eq:qosu} prevents a high-rate user from masking starvation of the other,
consistent with per-device QoS modeling in~\cite{zou2023ml}. The sensing floor
is imposed separately by~\eqref{eq:qoss}; the phase parameterization enforces
unit modulus automatically.

Problem~(P1) is non-convex because power and RIS phases are coupled. We enforce
QoS through penalties, while the softmax output satisfies the power simplex.

\section{Proposed MAML-Based Solution}
\subsection{Motivation for Meta-Learning}
The three power coefficients are low dimensional and reusable across related
channels, whereas the $N$ RIS phases require channel-specific adaptation.
Rather than solving the full non-convex problem from scratch or learning a
one-shot high-dimensional phase map, MAML learns a reusable outer model and
adapts the RIS with a fixed number of inner steps~\cite{maml,hospedales2022meta}.
This follows RIS-NOMA meta-learning~\cite{zou2023ml}; the implemented
first-order variant avoids the full higher-order graph
\cite{fallah2020maml,ji2022maml}.

\subsection{Feature Map and Power-Allocation Network}\subsection{Feature Map and Power-Allocation Network}\label{sec:feat}
The nine-dimensional feature $\bm z$ contains six effective gains (in dB),
communication/sensing QoS levels, and the SNR budget. A
9--256--256--256--3 LayerNorm/LeakyReLU network produces
\begin{equation}
\bm a=\operatorname{softmax}(f_{\bm\theta}(\bm z)),
\end{equation}
which enforces $a_n+a_f+a_T=1$ and $a_k\geq0$ by construction.

\subsection{Joint Power-Allocation and Phase Adaptation}
For a channel sample and the current phase vector $\bm\phi^{(j)}$, the network
first predicts
\begin{equation}
\bm a^{(j)}=\operatorname{softmax}
\left(f_{\bm\theta}(\bm z(\bm\phi^{(j)}))\right). \label{eq:powerpred}
\end{equation}
The implemented constrained loss is
\begin{align}
\ell^{(j)}={}&-\Rdl
+\lambda_c\big(R_{\mathrm{th},c}-R_n-R_f\big)_+
+\lambda_s\big(R_{\mathrm{th},s}-R_s\big)_+ \notag\\
&+\lambda_n\big(R_{\mathrm{th},n}-R_n\big)_+
+\lambda_f\big(R_{\mathrm{th},f}-R_f\big)_+, \label{eq:sampleloss}
\end{align}
where $(x)_+=\max(x,0)$. Here $\lambda_c,\lambda_s,\lambda_n,\lambda_f$ weight the cluster, sensing,
near-user, and far-user QoS penalties. With inner index $j$, $K$ phase steps,
inner rate $\eta_\phi$, outer rate $\alpha$, batch size $B$, and network
parameters $\bm\theta$, the updates are as follows.

Using $\theta_i=e^{j\phi_i}$ enforces unit modulus. Starting from a random
phase vector, Adam performs $K$ inner updates on~\eqref{eq:sampleloss},
\begin{equation}
\bm\phi^{(j+1)}=\operatorname{Adam}_{\phi}
\left(\bm\phi^{(j)},\nabla_{\bm\phi^{(j)}}\ell^{(j)};\eta_{\phi}\right),
\label{eq:inner}
\end{equation}
and the features/power split are recomputed after each phase update.

\subsection{First-Order Meta-Update}
After $K$ inner steps, the first-order MAML update treats
$\bm\phi^{(K)}$ as fixed and updates only the power network:
\begin{equation}
\bm\theta\leftarrow\bm\theta-\alpha
\nabla_{\bm\theta}\frac{1}{B}\sum_{b=1}^{B}
\ell_b\big(\bm\theta,\bm\phi_b^{(K)}\big), \label{eq:outer}
\end{equation}
The gradient is evaluated on phase-adapted channels; omitting derivatives
through the inner optimizer reduces memory and avoids the non-finite
higher-order meta-gradients observed in the implemented high-SINR setting.

\subsection{Training and Prediction Procedure}
Algorithm~\ref{alg:maml} summarizes training. At inference, the same $K$ phase
steps are applied to an unseen channel before the final softmax power split.

\begin{algorithm}[t]
\scriptsize
\SetAlgoLined
\DontPrintSemicolon
\KwIn{channels, $K$, $\eta_\phi$, $\alpha$, QoS penalties.}
\KwOut{$\bm\theta^\star$.}
Initialize $\bm\theta$\;
\While{training}{
 Sample $B$ channels\;
 \ForEach{channel $b$}{
   Initialize $\bm\phi_b^{(0)}$\;
   \For{$j=0$ \KwTo $K-1$}{
     Compute gains/features, predict $\bm a_b^{(j)}$, evaluate $\ell_b^{(j)}$\;
     Update $\bm\phi_b^{(j+1)}$ by inner Adam\;
   }
 }
 Update $\bm\theta$ using the first-order gradient in~\eqref{eq:outer}\;
}
Select the best validation checkpoint\;
\caption{First-order MAML-based power and RIS-phase optimization}
\label{alg:maml}
\end{algorithm}

\subsection{Convergence Analysis}
A global guarantee is not claimed because (P1) is non-convex, the hinge QoS
penalties are non-smooth at their boundaries, and the inner/outer variables
differ; standard smooth MAML results therefore do not directly apply
\cite{zou2023ml,fallah2020maml,ji2022maml}. Convergence is assessed empirically
from the fixed-step inner loop and the outer loss/rate plateau.

\subsection{Computational Complexity}\subsection{Computational Complexity}
Let $P_{\bm\theta}$ be the network parameter count
($\approx1.36\times10^5$). One inner step requires the network forward/backward
cost plus cascaded-channel and sensing-beam operations:
\begin{equation}
\mathcal O\!\left(P_{\bm\theta}+MN+M^3\right),
\end{equation}
and online prediction for one channel realization costs

\begin{equation}
\mathcal O\!\left(K(P_{\bm\theta}+MN+M^3)\right), \label{eq:onlinecomplex}
\end{equation}
and one training iteration scales as
$\mathcal O(BK(P_{\bm\theta}+MN+M^3))$. First-order MAML avoids storing the
full second-order graph through the $K$ inner steps.

\section{Conclusion}
We studied joint power allocation and RIS-phase design for MISO NOMA-ISAC.
The formulation combines cluster, individual-user, and sensing QoS constraints,
while a first-order MAML solver learns the power map and adapts RIS phases per
channel. Simulations show stable training and consistent gains from RIS
assistance and NOMA across the evaluated operating points.
